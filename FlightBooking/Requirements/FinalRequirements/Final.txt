    <script>
    (function () {
        'use strict';

        /* ---------- 0. Query parametreleri ---------- */
        const p = new URLSearchParams(window.location.search);
        const g = (k, d = '') => p.get(k) || d;

        const fromCity = g('fromCity');
        const toCity   = g('toCity');
        const fromIata = g('fromIata');
        const toIata   = g('toIata');
        const depart   = g('depart');
        const ret      = g('return');
        const adults   = parseInt(g('adults', '1'), 10);
        const children = parseInt(g('children', '0'), 10);
        const infants  = parseInt(g('infants', '0'), 10);
        const cabin    = g('cabin', 'economy');
        const currency = g('currency', 'TRY');
        const tripType = g('tripType', 'round');

        const cabinLabels = { economy:'Ekonomi', premium:'Premium Ekonomi', business:'Business', first:'First Class' };
        const tripLabels  = { round:'Gidiş-Dönüş', oneway:'Tek Yön', multi:'Çoklu Uçuş' };
        const aylar = ['Oca','Şub','Mar','Nis','May','Haz','Tem','Ağu','Eyl','Eki','Kas','Ara'];

        const esc = s => String(s ?? '').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');

        function fmtDate(iso) {
            if (!iso) return '';
            const d = new Date(iso + 'T00:00:00');
            if (isNaN(d)) return iso;
            return `${d.getDate()} ${aylar[d.getMonth()]} ${d.getFullYear()}`;
        }
        function paxText() {
            const parts = [];
            if (adults)   parts.push(`${adults} Yetişkin`);
            if (children) parts.push(`${children} Çocuk`);
            if (infants)  parts.push(`${infants} Bebek`);
            return parts.join(', ') || '1 Yetişkin';
        }

        /* ---------- Yardımcı: veri normalize ---------- */
        // "3 hr 50 min" -> 230 (dakika)
        function durationToMinutes(text) {
            if (!text) return 0;
            let h = 0, m = 0;
            const hMatch = text.match(/(\d+)\s*hr/);
            const mMatch = text.match(/(\d+)\s*min/);
            if (hMatch) h = parseInt(hMatch[1], 10);
            if (mMatch) m = parseInt(mMatch[1], 10);
            return h * 60 + m;
        }
        // "07:30 AM" -> 7.5 (ondalık saat, 0-24)
        function timeToHour(text) {
            if (!text) return -1;
            const m = text.match(/(\d{1,2}):(\d{2})\s*(AM|PM)?/i);
            if (!m) return -1;
            let h = parseInt(m[1], 10);
            const min = parseInt(m[2], 10);
            const ap = (m[3] || '').toUpperCase();
            if (ap === 'PM' && h < 12) h += 12;
            if (ap === 'AM' && h === 12) h = 0;
            return h + min / 60;
        }
        // fiyatı sayıya çevir; unavailable -> null
        function priceNum(price) {
            if (price === null || price === undefined) return null;
            const raw = price.toString().toLowerCase();
            if (raw === 'unavailable') return null;
            const n = Number(price);
            return isNaN(n) ? null : n;
        }

        // saat dilimi: sabah(5-11), öğle(11-17), akşam(17-22), gece(22-5)
        function timeSlot(hour) {
            if (hour < 0) return null;
            if (hour >= 5 && hour < 11)  return 'morning';
            if (hour >= 11 && hour < 17) return 'noon';
            if (hour >= 17 && hour < 22) return 'evening';
            return 'night';
        }

        /* ---------- 1. Özet çubuğu ---------- */
        const routeEl = document.getElementById('sfRoute');
        routeEl.innerHTML =
            `<span>${esc(fromCity)}${fromIata ? `<span class="sf-iata">${esc(fromIata)}</span>` : ''}</span>` +
            `<i class="bi bi-airplane"></i>` +
            `<span>${esc(toCity)}${toIata ? `<span class="sf-iata">${esc(toIata)}</span>` : ''}</span>`;

        let dateStr = fmtDate(depart);
        if (tripType === 'round' && ret) dateStr += ` – ${fmtDate(ret)}`;

        document.getElementById('sfMeta').innerHTML =
            `<span><i class="bi bi-calendar-event"></i> ${dateStr}</span>` +
            `<span><i class="bi bi-people"></i> ${paxText()}</span>` +
            `<span><i class="bi bi-briefcase"></i> ${cabinLabels[cabin] || 'Ekonomi'}</span>` +
            `<span><i class="bi bi-arrow-left-right"></i> ${tripLabels[tripType] || 'Gidiş-Dönüş'}</span>`;

        document.getElementById('sfResultTitle').textContent =
            `${fromCity || 'Kalkış'} – ${toCity || 'Varış'} Uçuşları`;

        /* ---------- 2. Kart HTML'i ---------- */
        function airlineLogo(f) {
            if (f.airlineLogo) {
                return `<img src="${esc(f.airlineLogo)}" alt="${esc(f.airline)}" class="sf-airline-logo"
                             onerror="this.outerHTML='<span class=\\'sf-airline-logo-fallback\\'><i class=\\'bi bi-airplane\\'></i></span>'">`;
            }
            return `<span class="sf-airline-logo-fallback"><i class="bi bi-airplane"></i></span>`;
        }
        function stopsBadge(f) {
            if (f.stops === 0) return `<span class="sf-stops sf-stops-direct">Direkt</span>`;
            const cls = f.stops === 1 ? 'sf-stops-1' : 'sf-stops-more';
            const layo = (f.layoverCities && f.layoverCities.length)
                ? `<span class="sf-layover">${esc(f.layoverCities.join(', '))}</span>` : '';
            return `<span class="sf-stops ${cls}">${f.stops} Aktarma</span>${layo}`;
        }
        function priceBlock(f) {
            const n = priceNum(f.price);
            if (n === null) return `<div class="sf-price-unavailable">Fiyat bulunamadı</div>`;
            return `
                <div class="sf-price">${n.toLocaleString('tr-TR')} <span class="sf-price-cur">${esc(currency)}</span></div>
                <button class="sf-select-btn">Seç</button>`;
        }
        function cardHtml(f) {
            return `
                <article class="sf-card">
                    <div class="sf-airline">
                        ${airlineLogo(f)}
                        <div class="sf-airline-name">${esc(f.airline) || 'Havayolu'}</div>
                    </div>
                    <div class="sf-times">
                        <div class="sf-time-block">
                            <div class="sf-time">${esc(f.departureTime) || '--'}</div>
                            <div class="sf-time-airport">${esc(f.departureAirport)}</div>
                        </div>
                        <div class="sf-path">
                            <div class="sf-duration">${esc(f.durationText)}</div>
                            <div class="sf-path-line"></div>
                            ${stopsBadge(f)}
                        </div>
                        <div class="sf-time-block">
                            <div class="sf-time">${esc(f.arrivalTime) || '--'}</div>
                            <div class="sf-time-airport">${esc(f.arrivalAirport)}</div>
                        </div>
                    </div>
                    <div class="sf-price-block">${priceBlock(f)}</div>
                </article>`;
        }

        /* ---------- 3. Durum gösterimleri ---------- */
        const listEl  = document.getElementById('sfList');
        const countEl = document.getElementById('sfResultCount');

        function showSkeleton() {
            listEl.innerHTML = Array(5).fill('<div class="sf-skeleton"></div>').join('');
        }
        function showError(msg) {
            listEl.innerHTML = `
                <div class="sf-state">
                    <i class="bi bi-exclamation-triangle"></i>
                    <div>${esc(msg || 'Uçuşlar getirilirken bir hata oluştu.')}</div>
                    <button class="sf-retry-btn" id="sfRetry">Tekrar Dene</button>
                </div>`;
            const btn = document.getElementById('sfRetry');
            if (btn) btn.addEventListener('click', loadFlights);
        }
        function showEmpty(msg) {
            listEl.innerHTML = `
                <div class="sf-state">
                    <i class="bi bi-airplane"></i>
                    <div>${esc(msg || 'Bu arama için uygun uçuş bulunamadı.')}</div>
                </div>`;
        }

        /* ---------- 4. State (filtre/sıralama) ---------- */
        let allFlights = [];      // backend'den gelen ham liste (normalize edilmiş)
        let priceMin = 0, priceMax = 0;

        // aktif filtre durumu
        const state = {
            stops: new Set(),         // 'direct' | '1' | '2plus'
            airlines: new Set(),      // seçili havayolları
            maxPrice: null,           // slider değeri
            timeSlots: new Set(),     // 'morning'|'noon'|'evening'|'night'
            sort: 'cheapest'
        };

        /* ---------- 5. Filtre panelini oluştur ---------- */
        function buildFilters() {
            // havayolları ve sayıları
            const airlineCounts = {};
            allFlights.forEach(f => {
                const a = f.airline || 'Bilinmiyor';
                airlineCounts[a] = (airlineCounts[a] || 0) + 1;
            });
            const airlines = Object.keys(airlineCounts).sort();

            // aktarma sayıları
            const stopCounts = { direct: 0, one: 0, more: 0 };
            allFlights.forEach(f => {
                if (f.stops === 0) stopCounts.direct++;
                else if (f.stops === 1) stopCounts.one++;
                else stopCounts.more++;
            });

            const timeSlotLabels = {
                morning: { label: 'Sabah', sub: '05–11', icon: 'bi-sunrise' },
                noon:    { label: 'Öğle',  sub: '11–17', icon: 'bi-sun' },
                evening: { label: 'Akşam', sub: '17–22', icon: 'bi-sunset' },
                night:   { label: 'Gece',  sub: '22–05', icon: 'bi-moon-stars' }
            };

            const html = `
                <div class="sf-filters-head">
                    <span class="sf-filters-title">Filtreler</span>
                    <button class="sf-clear-btn" data-role="clear">Temizle</button>
                </div>

                <!-- Aktarma -->
                <div class="sf-fgroup">
                    <div class="sf-fgroup-title"><i class="bi bi-signpost-split"></i> Aktarma</div>
                    <label class="sf-check">
                        <input type="checkbox" data-filter="stops" value="direct">
                        <span class="sf-check-label">Direkt</span>
                        <span class="sf-check-count">${stopCounts.direct}</span>
                    </label>
                    <label class="sf-check">
                        <input type="checkbox" data-filter="stops" value="1">
                        <span class="sf-check-label">1 Aktarma</span>
                        <span class="sf-check-count">${stopCounts.one}</span>
                    </label>
                    <label class="sf-check">
                        <input type="checkbox" data-filter="stops" value="2plus">
                        <span class="sf-check-label">2+ Aktarma</span>
                        <span class="sf-check-count">${stopCounts.more}</span>
                    </label>
                </div>

                <!-- Kalkış saati -->
                <div class="sf-fgroup">
                    <div class="sf-fgroup-title"><i class="bi bi-clock"></i> Kalkış Saati</div>
                    <div class="sf-time-grid">
                        ${Object.entries(timeSlotLabels).map(([key, v]) => `
                            <div class="sf-time-chip" data-filter="time" data-value="${key}">
                                <i class="bi ${v.icon}"></i>
                                <span>${v.label}</span>
                                <small>${v.sub}</small>
                            </div>`).join('')}
                    </div>
                </div>

                <!-- Fiyat -->
                <div class="sf-fgroup">
                    <div class="sf-fgroup-title"><i class="bi bi-cash-coin"></i> Maksimum Fiyat</div>
                    <div class="sf-range-wrap">
                        <input type="range" class="sf-range" data-filter="price"
                               min="${priceMin}" max="${priceMax}" value="${priceMax}" step="1">
                        <div class="sf-range-vals">
                            <span>${priceMin.toLocaleString('tr-TR')} ${esc(currency)}</span>
                            <strong data-role="price-current">${priceMax.toLocaleString('tr-TR')} ${esc(currency)}</strong>
                        </div>
                    </div>
                </div>

                <!-- Havayolu -->
                <div class="sf-fgroup">
                    <div class="sf-fgroup-title"><i class="bi bi-airplane-engines"></i> Havayolu</div>
                    ${airlines.map(a => `
                        <label class="sf-check">
                            <input type="checkbox" data-filter="airline" value="${esc(a)}">
                            <span class="sf-check-label">${esc(a)}</span>
                            <span class="sf-check-count">${airlineCounts[a]}</span>

            // hem desktop hem mobil panele bas
            document.getElementById('sfFilters').innerHTML = html;
            document.getElementById('sfFiltersMobile').innerHTML = html;

            // event'leri her iki panele de bağla
            bindFilterEvents(document.getElementById('sfFilters'));
            bindFilterEvents(document.getElementById('sfFiltersMobile'));

        /* ---------- 6. Filtre event bağlama ---------- */
        function bindFilterEvents(root) {
            // checkbox'lar (stops + airline)
            root.querySelectorAll('input[type="checkbox"][data-filter]').forEach(cb => {
                cb.addEventListener('change', () => {
                    const type = cb.dataset.filter;   // 'stops' | 'airline'
                    const val = cb.value;
                    const set = type === 'stops' ? state.stops : state.airlines;
                    if (cb.checked) set.add(val); else set.delete(val);
                    syncCheckboxes(type, val, cb.checked);
                    applyFilters();
                });
            });

            // saat dilimi chip'leri
            root.querySelectorAll('.sf-time-chip').forEach(chip => {
                chip.addEventListener('click', () => {
                    const val = chip.dataset.value;
                    if (state.timeSlots.has(val)) state.timeSlots.delete(val);
                    else state.timeSlots.add(val);
                    syncTimeChips();
                    applyFilters();
                });
            });

            // fiyat slider
            const range = root.querySelector('input[data-filter="price"]');
            if (range) {
                range.addEventListener('input', () => {
                    state.maxPrice = Number(range.value);
                    syncPriceLabel(root, state.maxPrice);
                    applyFilters();
                });
            }

            // temizle
            const clearBtn = root.querySelector('[data-role="clear"]');
            if (clearBtn) clearBtn.addEventListener('click', clearFilters);
        }

        // iki panel arasında checkbox durumunu senkronla
        function syncCheckboxes(type, val, checked) {
            document.querySelectorAll(`input[data-filter="${type}"][value="${CSS.escape(val)}"]`)
                .forEach(el => el.checked = checked);
        }
        function syncTimeChips() {
            document.querySelectorAll('.sf-time-chip').forEach(chip => {
                chip.classList.toggle('active', state.timeSlots.has(chip.dataset.value));
            });
        }
        function syncPriceLabel(root, val) {
            document.querySelectorAll('input[data-filter="price"]').forEach(r => r.value = val);
            document.querySelectorAll('[data-role="price-current"]').forEach(el => {
                el.textContent = `${Number(val).toLocaleString('tr-TR')} ${currency}`;
            });
        }

        function clearFilters() {
            state.stops.clear();
            state.airlines.clear();
            state.timeSlots.clear();
            state.maxPrice = priceMax;
            // UI reset
            document.querySelectorAll('input[type="checkbox"][data-filter]').forEach(cb => cb.checked = false);
            syncTimeChips();
            syncPriceLabel(null, priceMax);
            applyFilters();
        }

        /* ---------- 7. Filtre + sıralama uygula ---------- */
        function applyFilters() {
            let list = allFlights.slice();

            // aktarma
            if (state.stops.size) {
                list = list.filter(f => {
                    if (f.stops === 0) return state.stops.has('direct');
                    if (f.stops === 1) return state.stops.has('1');
                    return state.stops.has('2plus');
                });
            }
            // havayolu
            if (state.airlines.size) {
                list = list.filter(f => state.airlines.has(f.airline || 'Bilinmiyor'));
            }
            // saat dilimi
            if (state.timeSlots.size) {
                list = list.filter(f => f._slot && state.timeSlots.has(f._slot));
            }
            // fiyat (slider). Fiyatı olmayanları da göster (null geçir).
            if (state.maxPrice !== null) {
                list = list.filter(f => f._price === null || f._price <= state.maxPrice);
            }

            // sıralama
            list.sort((a, b) => {
                switch (state.sort) {
                    case 'fastest':  return a._dur - b._dur;
                    case 'earliest': return a._depHour - b._depHour;
                    case 'latest':   return b._depHour - a._depHour;
                    case 'cheapest':
                    default: {
                        // fiyatı olmayan en sona
                        if (a._price === null) return 1;
                        if (b._price === null) return -1;
                        return a._price - b._price;
                    }
                }
            });

            renderList(list);
        }

        function renderList(list) {
            if (!list.length) {
                showEmpty('Seçtiğiniz filtrelere uygun uçuş bulunamadı.');
                countEl.textContent = `0 / ${allFlights.length} uçuş`;
                return;
            }
            listEl.innerHTML = list.map(cardHtml).join('');
            countEl.textContent = (list.length === allFlights.length)
                ? `${list.length} uçuş bulundu`
                : `${list.length} / ${allFlights.length} uçuş gösteriliyor`;
        }

        /* ---------- 8. Sıralama select ---------- */
        document.getElementById('sfSort').addEventListener('change', e => {
            state.sort = e.target.value;
            applyFilters();
        });

        /* ---------- 9. Mobil drawer ---------- */
        const backdrop = document.getElementById('sfBackdrop');
        const drawer   = document.getElementById('sfDrawer');
        function openDrawer() { backdrop.classList.add('open'); drawer.classList.add('open'); }
        function closeDrawer() { backdrop.classList.remove('open'); drawer.classList.remove('open'); }
        document.getElementById('sfMobileFilterBtn').addEventListener('click', openDrawer);
        document.getElementById('sfDrawerClose').addEventListener('click', closeDrawer);
        document.getElementById('sfDrawerApply').addEventListener('click', closeDrawer);
        backdrop.addEventListener('click', closeDrawer);

        /* ---------- 10. Backend'den uçuşları çek ---------- */
        async function loadFlights() {
            if (!fromCity || !toCity || !depart) {
                showError('Arama bilgileri eksik. Lütfen aramayı yeniden yapın.');
                return;
            }
            showSkeleton();
            countEl.textContent = '';

            const qs = new URLSearchParams({
                fromCity, toCity, depart,
                adults: String(adults), cabin, currency
            });

            try {
                const res = await fetch('/Flight/SearchFlights?' + qs.toString());
                if (!res.ok) throw new Error('Sunucu hatası (' + res.status + ')');
                const data = await res.json();

                if (!data.success) { showError(data.error || 'Uçuşlar getirilemedi.'); return; }

                // IATA rozetlerini backend'in kesin değeriyle güncelle
                if (data.fromIata) {
                    const iataEls = routeEl.querySelectorAll('.sf-iata');
                    // rozet yoksa ekle
                    if (!iataEls.length) {
                        const spans = routeEl.querySelectorAll('span');
                        if (spans[0]) spans[0].insertAdjacentHTML('beforeend', `<span class="sf-iata">${esc(data.fromIata)}</span>`);
                        if (spans[1]) spans[1].insertAdjacentHTML('beforeend', `<span class="sf-iata">${esc(data.toIata)}</span>`);
                    } else {
                        if (iataEls[0]) iataEls[0].textContent = data.fromIata;
                        if (iataEls[1]) iataEls[1].textContent = data.toIata;
                    }
                }

                const flights = data.flights || [];
                if (!flights.length) { showEmpty(); countEl.textContent = '0 uçuş'; return; }

                // normalize et: filtre/sıralama için ek alanlar
                allFlights = flights.map(f => {
                    const pn = priceNum(f.price);
                    const dh = timeToHour(f.departureTime);
                    return {
                        ...f,
                        _price: pn,
                        _dur: durationToMinutes(f.durationText),
                        _depHour: dh < 0 ? 999 : dh,
                        _slot: timeSlot(dh)
                    };
                });

                // fiyat aralığını belirle (null olmayanlardan)
                const prices = allFlights.map(f => f._price).filter(v => v !== null);
                priceMin = prices.length ? Math.floor(Math.min(...prices)) : 0;
                priceMax = prices.length ? Math.ceil(Math.max(...prices)) : 0;
                state.maxPrice = priceMax;

                buildFilters();
                applyFilters();

            } catch (err) {
                showError(err.message);
            }
        }

        /* ---------- 11. Başlat ---------- */
        loadFlights();

    })();
    </script>
</body>
</html>

using System.Text.Json;
using FlightBooking.Dtos.FlightSearchDtos;

namespace FlightBooking.Services.AirportServices.FlighSearchServices
{
    public class FlightSearchService : IFlightSearchService
    {
        private readonly HttpClient _client;

        public FlightSearchService(HttpClient client)
        {
            _client = client;
        }

        public async Task<List<FlightCardDto>> SearchAsync(
            string fromIata,
            string toIata,
            string outboundDate,
            int adults,
            string cabin,
            string currency)
        {
            var cards = new List<FlightCardDto>();

            if (string.IsNullOrWhiteSpace(fromIata) || string.IsNullOrWhiteSpace(toIata))
                return cards;

            var travelClass = MapCabin(cabin);

            var url =
                $"https://google-flights2.p.rapidapi.com/api/v1/searchFlights" +
                $"?departure_id={Uri.EscapeDataString(fromIata)}" +
                $"&arrival_id={Uri.EscapeDataString(toIata)}" +
                $"&outbound_date={Uri.EscapeDataString(outboundDate)}" +
                $"&travel_class={travelClass}" +
                $"&adults={adults}" +
                $"&show_hidden=1" +
                $"&currency={Uri.EscapeDataString(currency)}" +
                $"&language_code=en-US&country_code=US&search_type=best";

            var request = new HttpRequestMessage
            {
                Method = HttpMethod.Get,
                RequestUri = new Uri(url),
                Headers =
                {
                    { "x-rapidapi-key", "630ce9cc86msh271c60cffe62d5ep1b514djsn0fe292593744" },
                    { "x-rapidapi-host", "google-flights2.p.rapidapi.com" },
                },
            };

            using var response = await _client.SendAsync(request);
            response.EnsureSuccessStatusCode();
            var body = await response.Content.ReadAsStringAsync();

            var parsed = JsonSerializer.Deserialize<FlightApiResponse>(body);
            var itin = parsed?.Data?.Itineraries;
            if (itin == null) return cards;

            var all = new List<FlightApiItinerary>();
            if (itin.TopFlights != null) all.AddRange(itin.TopFlights);
            if (itin.OtherFlights != null) all.AddRange(itin.OtherFlights);

            foreach (var it in all)
            {
                var firstLeg = it.Flights?.FirstOrDefault();
                var lastLeg = it.Flights?.LastOrDefault();

                var card = new FlightCardDto
                {
                    // ---- özet ----
                    Airline = firstLeg?.Airline ?? "",
                    AirlineLogo = it.AirlineLogo ?? firstLeg?.AirlineLogo ?? "",
                    DepartureTime = ExtractClock(it.DepartureTime),
                    ArrivalTime = ExtractClock(it.ArrivalTime),
                    DepartureAirport = firstLeg?.DepartureAirport?.AirportCode ?? fromIata,
                    ArrivalAirport = lastLeg?.ArrivalAirport?.AirportCode ?? toIata,
                    DurationText = it.Duration?.Text ?? "",
                    Stops = it.Stops,
                    LayoverCities = it.Layovers?
                        .Where(l => !string.IsNullOrWhiteSpace(l.City))
                        .Select(l => l.City!)
                        .ToList() ?? new List<string>(),
                    Price = NormalizePrice(it.Price),

                    // ---- detay: segmentler ----
                    Segments = it.Flights?.Select(leg => new FlightSegmentDto
                    {
                        Airline = leg.Airline ?? "",
                        AirlineLogo = leg.AirlineLogo ?? "",
                        FlightNumber = leg.FlightNumber ?? "",
                        Aircraft = leg.Aircraft ?? "",
                        Legroom = leg.Legroom ?? "",
                        DurationText = leg.Duration?.Text ?? "",
                        DepartureTime = leg.DepartureAirport?.Time ?? "",
                        DepartureCode = leg.DepartureAirport?.AirportCode ?? "",
                        DepartureName = leg.DepartureAirport?.AirportName ?? "",
                        ArrivalTime = leg.ArrivalAirport?.Time ?? "",
                        ArrivalCode = leg.ArrivalAirport?.AirportCode ?? "",
                        ArrivalName = leg.ArrivalAirport?.AirportName ?? ""
                    }).ToList() ?? new List<FlightSegmentDto>(),

                    // ---- detay: aktarmalar ----
                    Layovers = it.Layovers?.Select(l => new LayoverDto
                    {
                        AirportCode = l.AirportCode ?? "",
                        AirportName = l.AirportName ?? "",
                        City = l.City ?? "",
                        DurationLabel = l.DurationLabel ?? ""
                    }).ToList() ?? new List<LayoverDto>(),

                    // ---- detay: bagaj ----
                    Bags = it.Bags == null ? null : new BagsDto
                    {
                        CarryOn = it.Bags.CarryOn,
                        Checked = it.Bags.Checked
                    },

                    // ---- detay: karbon (gram -> kg) ----
                    Carbon = it.CarbonEmissions == null ? null : new CarbonDto
                    {
                        Co2eKg = it.CarbonEmissions.Co2e / 1000,
                        DifferencePercent = it.CarbonEmissions.DifferencePercent
                    }
                };

                cards.Add(card);
            }

            return cards;
        }

        // "25-08-2026 07:30 AM" -> "07:30 AM"
        private static string ExtractClock(string? raw)
        {
            if (string.IsNullOrWhiteSpace(raw)) return "";
            var parts = raw.Split(' ', 2);
            return parts.Length == 2 ? parts[1] : raw;
        }

        private static string NormalizePrice(JsonElement price)
        {
            return price.ValueKind switch
            {
                JsonValueKind.Number => price.GetRawText(),
                JsonValueKind.String => price.GetString() ?? "unavailable",
                _ => "unavailable"
            };
        }

        private static string MapCabin(string cabin) => cabin?.ToLower() switch
        {
            "business" => "BUSINESS",
            "first" => "FIRST",
            "premium" => "PREMIUM_ECONOMY",
            _ => "ECONOMY"
        };
    }
